# Routes the SDK/CLI/terraform testing contract does not apply to. Every entry
# carries its reason on the line directly above it, which
# scripts/check-simulator-tests.sh enforces: an exemption whose reason nobody
# wrote is indistinguishable from a test nobody wrote.
#
# The Compute Engine entries are all one situation: the meta-resource registrar
# mounts each collection at a path built from the collection's own name, so the
# contract reader — which resolves literal routes — records the scope prefix
# they share instead of the concrete routes. Every collection mounted under
# these prefixes carries its own SDK test; it is the prefix, not the surface,
# that has no test to point at.

# The aggregated list each collection declares, mounted as
# "/aggregated/" + collection.
GET /compute/v1/projects/{project}/aggregated/
# The global collections, mounted as "/global/" + collection.
GET /compute/v1/projects/{project}/global/
# The regional collections, mounted as "/regions/{region}/" + collection.
GET /compute/v1/projects/{project}/regions/{region}/
# The zonal collections, mounted as "/zones/{zone}/" + collection.
GET /compute/v1/projects/{project}/zones/{zone}/

# A Memorystore ACL policy's revision history, which the generated Google client
# does not wrap — the simulator's own custom-methods test drives both routes
# over HTTP the way a client would.
GET /v1/projects/{project}/locations/{location}/aclPolicies/{id}/revisions
# One revision of that history, addressed by the id the list returns.
GET /v1/projects/{project}/locations/{location}/aclPolicies/{id}/revisions/{revision}

# The one route serving both an image's family lookup and its IAM policy read,
# which a path router cannot separate. Its tail is a parameter, so the contract
# reader has no method name to match on; both documented methods behind it are
# exercised by TestCompute_ImageFamilyAndIamShareARouteWithoutShadowingEachOther.
GET /compute/v1/projects/{project}/global/images/{first}/{second}

# Enrolling a project in a preview feature. The route's tail is a parameter, so
# the contract reader has no method name to match on; the route answers 501 —
# the set of preview features is Google's own — and
# TestCompute_GooglePublishedCatalogsAreDeclaredGaps asserts it says so.
PATCH /compute/v1/projects/{project}/global/previewFeatures/{previewFeature}

# A zonal disk's patch. Compute Engine declares the method and the simulator
# serves it, but google-api-go-client's DisksService wraps no Patch, so no SDK
# test can reach it — the CLI and the terraform provider go through update and
# resize instead.
PATCH /compute/v1/projects/{project}/zones/{zone}/disks/{name}

# What a composite health check has observed. Compute Engine declares the read
# and the simulator serves it, but google-api-go-client wraps no
# CompositeHealthChecks, so no SDK test can reach it.
GET /compute/v1/projects/{project}/regions/{region}/compositeHealthChecks/{name}/getHealth
# The same for a health source, whose collection the generated client also does
# not wrap.
GET /compute/v1/projects/{project}/regions/{region}/healthSources/{name}/getHealth

# Deleting a key ring. Cloud KMS declares the method and the simulator serves
# it — refusing a ring that still holds keys, because Cloud KMS never removes
# key material as a side effect of removing its container — but the vendored
# google-api-go-client predates the method and its KeyRingsService wraps no
# Delete, so no SDK test can reach it.
DELETE /v1/projects/{project}/locations/{location}/keyRings/{keyRing}

# Cloud Billing's conversational agent task surface, added in the 2026-09-04
# spec refresh and declared unserved in cloudbilling.go. The vendored
# google-api-go-client (google.golang.org/api/cloudbilling/v1) predates all
# ten of these methods and wraps none of them, so no SDK test can reach any
# of them — not even to assert the 501. tasks.get / tasks.subscribe, the
# {tasksAction} colon-verb spelling.
GET /v1/tasks/{tasksAction}
# tasks.cancel, the {tasksAction} colon-verb spelling.
POST /v1/tasks/{tasksAction}
# tasks.pushNotificationConfigs.list.
GET /v1/tasks/{tasksId}/pushNotificationConfigs
# tasks.pushNotificationConfigs.create.
POST /v1/tasks/{tasksId}/pushNotificationConfigs
# tasks.pushNotificationConfigs.get.
GET /v1/tasks/{tasksId}/pushNotificationConfigs/{pushNotificationConfigsId}
# tasks.pushNotificationConfigs.delete.
DELETE /v1/tasks/{tasksId}/pushNotificationConfigs/{pushNotificationConfigsId}
# v1.getCard, the payment card read.
GET /v1/card
# message.send.
POST /v1/message:send
# message.stream.
POST /v1/message:stream
# The generic AIP-127 resource-name spelling of tasks.get /
# tasks.pushNotificationConfigs.get ({+name}); same SDK gap, and the contract
# reader has no method name to match a wildcard tail against regardless.
GET /v1/tasks/{rest...}
# The generic AIP-127 resource-name spelling of
# tasks.pushNotificationConfigs.delete ({+name}); same reasoning.
DELETE /v1/tasks/{rest...}

# Compute Engine's regional project view, added in the same refresh and
# declared unserved in compute_catalogs.go: the vendored google-api-go-client's
# RegionsService wraps no ProjectViews method.
GET /compute/v1/projects/{project}/regions/{region}/projectViews
# regions.advice.capacity: AdviceService wraps only CalendarMode, not
# Capacity — the client library has not caught up to the spec either.
POST /compute/v1/projects/{project}/regions/{region}/advice/capacity
# regions.advice.capacityHistory: same AdviceService gap.
POST /compute/v1/projects/{project}/regions/{region}/advice/capacityHistory
